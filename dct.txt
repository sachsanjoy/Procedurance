import pandas as pd

# Sample Data
data = {'date': ['2024-01-20', '2024-01-25', '2024-01-27',
                 '2024-02-24', '2024-02-26',
                 '2024-03-23', '2024-03-28'],
        'metricvalue': [100, 200, 300, 150, 250, 180, 280]}

# Create DataFrame
df = pd.DataFrame(data)

# Convert date column to datetime
df['date'] = pd.to_datetime(df['date'])

# Function to get closest date ≤ 25th for each month
def closest_date(group):
    return group[group['date'] <= group['date'].dt.to_period('M').dt.to_timestamp() + pd.DateOffset(days=24)].max()

# Apply filtering per year-month group
filtered_df = df.groupby(df['date'].dt.to_period('M')).apply(closest_date).reset_index(drop=True)

# Display result
print(filtered_df)
