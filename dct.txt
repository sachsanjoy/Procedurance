    "import numpy as np\n",
    "import pandas as pd\n",
    "from sklearn.cluster import DBSCAN\n",
    "\n",
    "def generate_synthetic_data(end_date_str=\"2026-05-31\", periods=60):\n",
    "    \"\"\"\n",
    "    Generates a DataFrame with a monthly Date column and 50 synthetic time series.\n",
    "    Each series contains patterns (sine, trends, or random noise) and has \n",
    "    randomized start/end timelines (filled with NaN).\n",
    "    \"\"\"\n",
    "    np.random.seed(42)\n",
    "    \n",
    "    # 1. Create monthly date range ending at the specified date\n",
    "    date_range = pd.date_range(end=end_date_str, periods=periods, freq='ME')\n",
    "    df = pd.DataFrame({'Date': date_range})\n",
    "    \n",
    "    x = np.linspace(0, 4 * np.pi, periods)\n",
    "    \n",
    "    # 2. Generate 50 distinct columns with various patterns\n",
    "    for i in range(1, 51):\n",
    "        col_name = f\"Series_{i}\"\n",
    "        \n",
    "        # Mix up patterns so some actually cluster together\n",
    "        if i % 3 == 0:\n",
    "            # Sine wave pattern\n",
    "            df[col_name] = np.sin(x) + np.random.normal(0, 0.1, periods)\n",
    "        elif i % 3 == 1:\n",
    "            # Upward trend pattern\n",
    "            df[col_name] = np.linspace(0, 2, periods) + np.random.normal(0, 0.1, periods)\n",
    "        else:\n",
    "            # Pure noise (should end up as DBSCAN outliers / -1)\n",
    "            df[col_name] = np.random.normal(0, 1, periods)\n",
    "            \n",
    "        # 3. Simulate different start/end timelines by injecting NaNs\n",
    "        # Randomly clip the beginning and ending of each series\n",
    "        start_clip = np.random.randint(0, int(periods * 0.25))\n",
    "        end_clip = np.random.randint(int(periods * 0.75), periods + 1)\n",
    "        \n",
    "        # Apply the masks\n",
    "        df.loc[df.index < start_clip, col_name] = np.nan\n",
    "        df.loc[df.index >= end_clip, col_name] = np.nan\n",
    "        \n",
    "    return df\n",
    "\n",
    "\n",
    "def process_timeline_window(df, start_date, end_date, min_data_pct=0.7):\n",
    "    \"\"\"\n",
    "    Slices the dataframe to the user's selected timeline and filters/imputes data.\n",
    "    \n",
    "    Parameters:\n",
    "    - min_data_pct: Drop columns that are missing more than X% of data in this window.\n",
    "    \"\"\"\n",
    "    # Ensure Date is index for easy slicing\n",
    "    df_temp = df.set_index('Date') if 'Date' in df.columns else df.copy()\n",
    "    \n",
    "    # Slice to timeline\n",
    "    window_df = df_temp.loc[start_date:end_date]\n",
    "    \n",
    "    if window_df.empty:\n",
    "        raise ValueError(\"The selected timeline window contains no data.\")\n",
    "        \n",
    "    # Drop columns that don't have enough data points in this specific window\n",
    "    min_required_points = int(len(window_df) * min_data_pct)\n",
    "    window_df = window_df.dropna(thresh=min_required_points, axis=1)\n",
    "    \n",
    "    # Forward-fill and backward-fill remaining minor gaps (required for correlation)\n",
    "    window_df = window_df.ffill().bfill()\n",
    "    \n",
    "    return window_df\n",
    "\n",
    "\n",
    "def compute_distance_matrix(window_df, metric_type='correlation'):\n",
    "    \"\"\"\n",
    "    Computes an NxN distance matrix from the aligned time series.\n",
    "    Identical series = 0, highly dissimilar series approach 2.\n",
    "    \"\"\"\n",
    "    if metric_type == 'correlation':\n",
    "        # Pearson correlation matrix (ranges from -1 to 1)\n",
    "        similarity_matrix = window_df.corr(method='pearson')\n",
    "        distance_matrix = 1 - similarity_matrix\n",
    "        \n",
    "    elif metric_type == 'cosine':\n",
    "        # Normalize data (Z-score) first because cosine is scale/mean sensitive\n",
    "        normalized_df = (window_df - window_df.mean()) / window_df.std()\n",
    "        \n",
    "        # Compute Cosine Similarity\n",
    "        # Dot product of normalized matrix divided by number of features\n",
    "        sim_matrix = np.dot(normalized_df.T, normalized_df) / (len(normalized_df) - 1)\n",
    "        similarity_matrix = pd.DataFrame(sim_matrix, index=window_df.columns, columns=window_df.columns)\n",
    "        distance_matrix = 1 - similarity_matrix\n",
    "    else:\n",
    "        raise ValueError(\"metric_type must be 'correlation' or 'cosine'\")\n",
    "        \n",
    "    # Clean up minor floating-point anomalies\n",
    "    np.fill_diagonal(distance_matrix.values, 0.0)\n",
    "    distance_matrix = distance_matrix.clip(0, 2)\n",
    "    \n",
    "    return distance_matrix\n",
    "\n",
    "\n",
    "def cluster_with_dbscan(distance_matrix, eps=0.4, min_samples=3):\n",
    "    \"\"\"\n",
    "    Applies DBSCAN using a precomputed distance matrix.\n",
    "    Returns a dictionary mapping series names to cluster labels.\n",
    "    \"\"\"\n",
    "    # Run DBSCAN\n",
    "    dbscan = DBSCAN(eps=eps, min_samples=min_samples, metric='precomputed')\n",
    "    labels = dbscan.fit_predict(distance_matrix.values)\n",
    "    \n",
    "    # Map results back to series names\n",
    "    results = pd.Series(labels, index=distance_matrix.index, name='Cluster')\n",
    "    \n",
    "    return results.to_frame()"

    "    n = dist_matrix.shape[0]\n",
